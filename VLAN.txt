Switch> enable
Switch# configure terminal

! --- Create VLANs ---
Switch(config)# vlan 150
Switch(config-vlan)# name Admin
Switch(config-vlan)# exit

Switch(config)# vlan 200
Switch(config-vlan)# name Students
Switch(config-vlan)# exit

! --- Configure Access Ports (Example: FastEthernet0/1 for Admin) ---
Switch(config)# interface FastEthernet0/1
Switch(config-if)# switchport mode access
Switch(config-if)# switchport access vlan 150
Switch(config-if)# exit

! --- Configure Access Ports (Example: FastEthernet0/2 for Students) ---
Switch(config)# interface FastEthernet0/2
Switch(config-if)# switchport mode access
Switch(config-if)# switchport access vlan 200
Switch(config-if)# exit

! --- Configure Trunk Port to Switch 2 (GigabitEthernet0/1) ---
Switch(config)# interface GigabitEthernet0/1
Switch(config-if)# switchport trunk encapsulation dot1q  ! (Required on older Catalyst switches)
Switch(config-if)# switchport mode trunk
Switch(config-if)# switchport trunk allowed vlan 150,200
Switch(config-if)# end
